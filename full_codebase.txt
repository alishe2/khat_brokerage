class AuthGate extends StatelessWidget {
  const AuthGate({super.key});

  @override
  Widget build(BuildContext context) {
    final supabase = Supabase.instance.client;

    return StreamBuilder<AuthState>(
      stream: supabase.auth.onAuthStateChange,

      builder: (context, snapshot) {
        // ==========================================================
        // NO LOGGED-IN USER
        // ==========================================================

        final session = supabase.auth.currentSession;

        if (session == null) {
          return const LoginScreen();
        }

        // ==========================================================
        // USER IS LOGGED IN
        // ==========================================================

        return const _RoleGate();
      },
    );
  }
}

class _RoleGateState extends State<_RoleGate> {
  final _supabase = Supabase.instance.client;

  String? _error;

  bool _isLoading = true;

  @override
  void initState() {
    super.initState();

    _loadRole();
  }

  // ==============================================================
  // LOAD CURRENT USER ROLE
  // ==============================================================

  Future<void> _loadRole() async {
    try {
      final role = await _supabase.rpc(
        'get_my_role',
      );

      if (!mounted) return;

      if (role == null || role.toString().isEmpty) {
        setState(() {
          _error = 'User role was not found.';
          _isLoading = false;
        });

        return;
      }

      setState(() {
        _role = role.toString();
        _isLoading = false;
      });
    } catch (e) {
      if (!mounted) return;

      setState(() {
        _error = 'Could not load user role: $e';
        _isLoading = false;
      });
    }
  }

  // ==============================================================
  // BUILD
  // ==============================================================

  @override
  Widget build(BuildContext context) {
    // ============================================================
    // LOADING
    // ============================================================

    if (_isLoading) {
      return const Scaffold(
        body: Center(
          child: CircularProgressIndicator(),
        ),
      );
    }

    // ============================================================
    // ERROR
    // ============================================================

    if (_error != null) {
      return Scaffold(
        body: Center(
          child: Padding(

  final _formKey = GlobalKey<FormState>();

  final _phoneController = TextEditingController();
  final _passwordController = TextEditingController();

  bool _obscurePassword = true;
  bool _isLoading = false;

    _phoneController.dispose();
    _passwordController.dispose();
    super.dispose();
  }

  Future<void> _login() async {
    if (!_formKey.currentState!.validate()) {
      return;
    }

    setState(() {
      _isLoading = true;
    });

    try {
      final phone = _phoneController.text.trim();
      final password = _passwordController.text;

      // Find the internal authentication email using the phone number.
      final authEmail = await Supabase.instance.client.rpc(
        'get_auth_email_by_phone',
        params: {'p_phone': phone},
      );

      if (authEmail == null || authEmail.toString().isEmpty) {
        throw const AuthException('Invalid phone number or password.');

    if (_dateSearch.isEmpty) {
      return true;
    }

    if (dateValue == null || dateValue.isEmpty) {
      return false;
    }

    final search = _normalizeDate(_dateSearch);

    final date = dateValue.length >= 10
        ? dateValue.substring(0, 10)
        : dateValue;

    // Full date: 2026-08-30
    if (RegExp(r'^\d{4}-\d{2}-\d{2}$').hasMatch(search)) {
      return date == search;
    }

    // Month: 2026-08
    if (RegExp(r'^\d{4}-\d{2}$').hasMatch(search)) {
      return date.startsWith(search);
    }

    // Year: 2026
    if (RegExp(r'^\d{4}$').hasMatch(search)) {
      return date.startsWith(search);
    }

    return date.contains(search);
  }

  Future<void> _loadDashboard() async {

    try {
      final today = DateTime.now();

      final todayString =

          '${today.day.toString().padLeft(2, '0')}';

      final userId = _supabase.auth.currentUser?.id;

      final results = await Future.wait<dynamic>([
        userId != null

            .eq('sales.sale_date', todayString),

            .eq('payment_date', todayString),

        _supabase

      ]);

      final userData = results[0] as Map<String, dynamic>?;
      final salesData = results[1] as List;
      final saleBuyerData = results[2] as List;
      final oldDebtPaymentsData = results[3] as List;
      final withdrawalsData = results[4] as List;

      String accountantName = 'Accountant';

      if (userId != null) {
        accountantName = userData?['full_name']?.toString().trim() ?? '';

        if (accountantName.isEmpty) {
          accountantName = 'Accountant';
        }
      }

      int kiloSold = 0;
      double commission = 0;

      for (final sale in salesData) {
        kiloSold += (sale['total_bundles'] as num?)?.toInt() ?? 0;
        commission += (sale['commission_amount'] as num?)?.toDouble() ?? 0;
      }

      double buyerPaid = 0;
      double buyerUnpaid = 0;

      for (final buyer in saleBuyerData) {
        buyerPaid += (buyer['paid_amount'] as num?)?.toDouble() ?? 0;
        buyerUnpaid += (buyer['unpaid_amount'] as num?)?.toDouble() ?? 0;
      }

      double paidFromOldDebt = 0;

      for (final payment in oldDebtPaymentsData) {
        paidFromOldDebt += (payment['amount'] as num?)?.toDouble() ?? 0;

      }

        _buyerUnpaidToday = buyerUnpaid;

  @override

  Future<void> _loadDashboard() async {
    if (mounted) {
      setState(() {
        _isLoading = true;
      });
    }

    try {
      final today = DateTime.now();

      final todayString =
          '${today.year.toString().padLeft(4, '0')}-'
          '${today.month.toString().padLeft(2, '0')}-'
          '${today.day.toString().padLeft(2, '0')}';

      final userId = _supabase.auth.currentUser?.id;

      // ============================================================
      // FIRE ALL NINE QUERIES AT ONCE
      //
      // None of these depend on each other's results â€” only on
      // `todayString` / `userId`, both already known above. Instead
      // of nine sequential round trips, we start them all at once
      // and wait for the slowest one, not the sum of all nine.
      // ============================================================

      final results = await Future.wait<dynamic>([
        // 0: admin name â€” placeholder keeps list positions stable
        // even when there's no userId to query with.
        userId != null
            ? _supabase
                  .from('users')
                  .select('full_name')
                  .eq('id', userId)
                  .maybeSingle()
            : Future.value(null),

        // 1: today's sales
        _supabase
            .from('sales')
            .select('total_bundles, commission_amount')
            .eq('sale_date', todayString),

        // 2: today's buyer payments from today's sales
        _supabase
            .from('sale_buyers')
            .select('paid_amount, unpaid_amount, sales!inner(sale_date)')
            .eq('sales.sale_date', todayString),

        // 3: paid from old debt
        _supabase
            .from('buyer_debt_payments')
            .select('amount')
            .eq('payment_date', todayString),

        // 4: total farmer money
        _supabase
            .from('farmers')
            .select('current_balance')
            .eq('is_active', true),

        // 5: total buyer debt
        _supabase.from('buyers').select('current_debt').eq('is_active', true),

        // 6: today's farmer withdrawals
        _supabase
            .from('farmer_withdrawals')
            .select('amount')
            .eq('withdrawal_date', todayString),

        // 7: today's worker fees
        _supabase
            .from('worker_fees')
            .select('amount')
            .eq('fee_date', todayString)
            .maybeSingle(),

        // 8: today's other expenses
        _supabase
            .from('other_expenses')
            .select('amount')
            .eq('expense_date', todayString),
      ]);

      // ============================================================
      // UNPACK RESULTS â€” same order the futures were listed above
      // ============================================================

      final userData = results[0] as Map<String, dynamic>?;
      final salesData = results[1] as List;
      final saleBuyerData = results[2] as List;
      final oldDebtPaymentsData = results[3] as List;
      final farmersData = results[4] as List;
      final buyersData = results[5] as List;
      final withdrawalsData = results[6] as List;
      final workerFeeData = results[7] as Map<String, dynamic>?;
      final otherExpensesData = results[8] as List;

      // ============================================================
      // ADMIN NAME
      // ============================================================

      String adminName = 'Administrator';

      if (userId != null) {
        adminName = userData?['full_name']?.toString().trim() ?? '';

          adminName = 'Administrator';
        }
      }

      // ============================================================
      // TODAY'S SALES
      // ============================================================

      int kiloSold = 0;
      double commission = 0;

      for (final sale in salesData) {
        kiloSold += (sale['total_bundles'] as num?)?.toInt() ?? 0;
        commission += (sale['commission_amount'] as num?)?.toDouble() ?? 0;
      }

      // ============================================================
      // TODAY'S BUYER PAYMENTS FROM TODAY'S SALES
      // ============================================================

      double buyerPaid = 0;
      double buyerUnpaid = 0;

      for (final buyer in saleBuyerData) {
        buyerPaid += (buyer['paid_amount'] as num?)?.toDouble() ?? 0;
        buyerUnpaid += (buyer['unpaid_amount'] as num?)?.toDouble() ?? 0;
      }

      // ============================================================
      // PAID FROM OLD DEBT
      // ============================================================

      double paidFromOldDebt = 0;

      for (final payment in oldDebtPaymentsData) {
        paidFromOldDebt += (payment['amount'] as num?)?.toDouble() ?? 0;
      }

      final totalPaid = buyerPaid + paidFromOldDebt;

      // ============================================================
      // TOTAL FARMER MONEY
      // ============================================================

      double farmerMoney = 0;

      for (final farmer in farmersData) {
        farmerMoney += (farmer['current_balance'] as num?)?.toDouble() ?? 0;
      }

      // ============================================================
      // TOTAL BUYER DEBT
      // ============================================================

      double buyerDebt = 0;

      for (final buyer in buyersData) {
        buyerDebt += (buyer['current_debt'] as num?)?.toDouble() ?? 0;
      }

      // ============================================================
      // TODAY'S FARMER WITHDRAWALS
      // ============================================================

      double farmerWithdrawals = 0;

      for (final withdrawal in withdrawalsData) {
        farmerWithdrawals += (withdrawal['amount'] as num?)?.toDouble() ?? 0;
      }

      // ============================================================
      // TODAY'S WORKER FEES
      // ============================================================

      double? workerFees;

      if (workerFeeData != null) {
        workerFees = (workerFeeData['amount'] as num?)?.toDouble() ?? 0;
      }

      // ============================================================
      // TODAY'S OTHER EXPENSES
      // ============================================================

      double otherExpenses = 0;

        _commissionToday = commission;

        _buyerUnpaidToday = buyerUnpaid;

        _totalPaidToday = totalPaid;

        _buyerDebt = buyerDebt;

        _workerFeesToday = workerFees ?? 0;
        _otherExpensesToday = otherExpenses;

        _adminProfitToday = adminProfit;

        _isLoading = false;
      });
    } catch (e) {
      if (!mounted) return;

      setState(() {
        _isLoading = false;
      });

      ScaffoldMessenger.of(
        context,
      ).showSnackBar(SnackBar(content: Text('Could not load dashboard: $e')));

class _FarmerTransactionHistoryState extends State<FarmerTransactionHistory> {
  final _supabase = Supabase.instance.client;

  bool _loading = true;

  List<Map<String, dynamic>> _sales = [];
  List<Map<String, dynamic>> _withdrawals = [];
  List<Map<String, dynamic>> _adjustments = [];

  // ================================================================
  // DATE SEARCH
  // ================================================================

  final TextEditingController _dateSearchController = TextEditingController();

  @override
  void initState() {
    super.initState();
    _loadHistory();
  }

  @override
  void dispose() {
    _dateSearchController.dispose();
    super.dispose();
  }

  @override
  void didUpdateWidget(covariant FarmerTransactionHistory oldWidget) {
    super.didUpdateWidget(oldWidget);

    if (oldWidget.refreshTrigger != widget.refreshTrigger ||
        oldWidget.farmerId.toString() != widget.farmerId.toString()) {
      _loadHistory();
    }
  }

  // ================================================================
  // LOAD EVERYTHING
  // ================================================================

  Future<void> _loadHistory() async {
    if (!mounted) return;

    setState(() {
      _loading = true;
    });

    try {
      final farmerId = widget.farmerId;

      // ============================================================
      // SALES
      // ============================================================

      final salesData = await _supabase
          .from('sales')
          .select('''
            id,
            transaction_code,
            sale_date,
            total_bundles,
            total_sales,
            commission_amount,
            advance_loan,
            farmer_amount,
            seller_id,
            sellers(full_name)
          ''')
          .eq('farmer_id', farmerId)
          .order('sale_date', ascending: false);

      final sales = List<Map<String, dynamic>>.from(salesData);

      // ============================================================
      // SALE IDs
      // ============================================================

      final saleIds = sales
          .map((sale) => sale['id'])
          .where((id) => id != null)
          .toList();

      // ============================================================
      // SALE BUYERS
      // ============================================================

      final Map<String, List<Map<String, dynamic>>> priceLinesBySale = {};

      if (saleIds.isNotEmpty) {
        final saleBuyersData = await _supabase
            .from('sale_buyers')
            .select('id, sale_id')
            .inFilter('sale_id', saleIds);

        final saleBuyers = List<Map<String, dynamic>>.from(saleBuyersData);

        // ============================================================
        // SALE BUYER IDs
        // ============================================================

        final saleBuyerIds = saleBuyers
            .map((buyer) => buyer['id'])
            .where((id) => id != null)
            .toList();

        // ============================================================
        // PRICE LINES
        // ============================================================

        if (saleBuyerIds.isNotEmpty) {
          final priceLinesData = await _supabase
              .from('sale_price_lines')
              .select('''
                id,
                sale_buyer_id,
                bundle_count,
                price_per_bundle,
                line_total
              ''')
              .inFilter('sale_buyer_id', saleBuyerIds);

          final priceLines = List<Map<String, dynamic>>.from(priceLinesData);

          // ==========================================================
          // CONNECT PRICE LINES â†’ SALE
          // ==========================================================

          for (final buyer in saleBuyers) {
            final saleId = buyer['sale_id']?.toString();
            final saleBuyerId = buyer['id']?.toString();

            if (saleId == null || saleBuyerId == null) {
              continue;
            }

            final buyerLines = priceLines
                .where(
                  (line) => line['sale_buyer_id']?.toString() == saleBuyerId,
                )
                .toList();

            if (buyerLines.isEmpty) {
              continue;
            }

            priceLinesBySale.putIfAbsent(saleId, () => []);

            priceLinesBySale[saleId]!.addAll(
              buyerLines.map((line) => Map<String, dynamic>.from(line)),
            );
          }
        }
      }

      // ============================================================
      // ATTACH PRICE LINES TO SALES
      // ============================================================

      for (final sale in sales) {
        final saleId = sale['id']?.toString();

        sale['price_lines'] = priceLinesBySale[saleId] ?? [];
      }

      // ============================================================
      // WITHDRAWALS
      // ============================================================

      final withdrawalsData = await _supabase
          .from('farmer_withdrawals')
          .select('''
            id,
            amount,
            withdrawal_date,
            remaining_balance
          ''')
          .eq('farmer_id', farmerId)
          .order('withdrawal_date', ascending: false);

      final withdrawals = List<Map<String, dynamic>>.from(withdrawalsData);

      // ============================================================
      // BALANCE ADJUSTMENTS
      // ============================================================

          .from('farmer_balance_adjustments')
          .select('''
            id,
            old_balance,
            new_balance,
            reason,
            created_at
          ''')
          .eq('farmer_id', farmerId)
          .order('created_at', ascending: false);

      final adjustments = List<Map<String, dynamic>>.from(adjustmentsData);

      if (!mounted) return;

      setState(() {
        _sales = sales;
        _withdrawals = withdrawals;
        _adjustments = adjustments;
        _loading = false;
      });
    } catch (e) {

      });

        ),

  String _normalizeDate(String value) {
    return value.trim().replaceAll('/', '-');
  }

  bool _matchesDateSearch(String? dateValue) {
    if (_dateSearch.isEmpty) {
      return true;
    }

    if (dateValue == null || dateValue.isEmpty) {
      return false;
    }

    final search = _normalizeDate(_dateSearch);
    final date = dateValue.length >= 10
        ? dateValue.substring(0, 10)
        : dateValue;

    // Full date: 2026-08-30
    if (RegExp(r'^\d{4}-\d{2}-\d{2}$').hasMatch(search)) {
      return date == search;
    }

    // Month: 2026-08
    if (RegExp(r'^\d{4}-\d{2}$').hasMatch(search)) {
      return date.startsWith(search);
    }

    // Year: 2026
    if (RegExp(r'^\d{4}$').hasMatch(search)) {
      return date.startsWith(search);
    }

    // Also allow partial searching if the user types something else.
    return date.contains(search);
  }

  List<Map<String, dynamic>> get _filteredSales {
    return _sales.where((sale) {
      return _matchesDateSearch(sale['sale_date']?.toString());
    }).toList();
  }

  List<Map<String, dynamic>> get _filteredWithdrawals {
    return _withdrawals.where((withdrawal) {
      return _matchesDateSearch(withdrawal['withdrawal_date']?.toString());
    }).toList();
  }

  List<Map<String, dynamic>> get _filteredAdjustments {
    return _adjustments.where((adjustment) {

    }).toList();
  }

    setState(() {
      _filteredItems = widget.items.where((item) {
        final name = (item[widget.nameKey] ?? '').toString().toLowerCase();

        final code = (item[widget.codeKey] ?? '').toString().toLowerCase();

            code.contains(search) ||
            phone.contains(search) ||

  final _amountController = TextEditingController();

  bool _isLoading = true;
  bool _isSaving = false;

  double _todayAmount = 0;

  String get _today {
    final now = DateTime.now();

    return '${now.year.toString().padLeft(4, '0')}-'
        '${now.month.toString().padLeft(2, '0')}-'
        '${now.day.toString().padLeft(2, '0')}';
  }

  @override
  void initState() {
    super.initState();
    _loadTodayFee();
  }

  @override
  void dispose() {
    _amountController.dispose();
    super.dispose();
  }

  Future<void> _loadTodayFee() async {
    try {
      final data = await _supabase
          .from('worker_fees')
          .select('amount')
          .eq('fee_date', _today)
          .maybeSingle();

      if (!mounted) return;

      final amount =
          (data?['amount'] as num?)?.toDouble() ?? 0;

      setState(() {
        _todayAmount = amount;

        _amountController.text =
            amount == 0 ? '' : amount.toStringAsFixed(2);

        _isLoading = false;
      });
    } catch (e) {
      if (!mounted) return;

      setState(() {
        _isLoading = false;
      });

      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          content: Text(
            'Could not load worker fees: $e',
          ),
        ),
      );
    }
  }

  Future<void> _saveFee() async {
    final text = _amountController.text.trim();

    final amount = double.tryParse(text);

    if (amount == null || amount < 0) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text(
            'Enter a valid worker fee amount',
          ),
        ),
      );

      return;
    }

    final userId = _supabase.auth.currentUser?.id;

    if (userId == null) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text('You are not logged in'),
        ),
      );

      return;
    }

    setState(() {
      _isSaving = true;
    });

    try {
      // Check whether today's record already exists.
      final existing = await _supabase
          .from('worker_fees')
          .select('id')
          .eq('fee_date', _today)
          .maybeSingle();

      if (existing != null) {
        // Update today's existing total.
        await _supabase
            .from('worker_fees')
            .update({
              'amount': amount,
            })
            .eq('id', existing['id']);
      } else {
        // Create today's fee record.
        await _supabase
            .from('worker_fees')
            .insert({
              'fee_date': _today,
              'amount': amount,
              'recorded_by': userId,
            });
      }

      if (!mounted) return;

      setState(() {
        _todayAmount = amount;
        _isSaving = false;
      });

      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text(
            'Worker fees saved successfully',
          ),
        ),
      );
    } catch (e) {
      if (!mounted) return;

      setState(() {
        _isSaving = false;
      });

      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          content: Text(
            'Could not save worker fees: $e',
          ),
        ),
      );
    }
  }
